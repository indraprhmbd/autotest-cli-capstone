Problem Statement

In fast-moving software development, unit tests are frequently skipped or written incompletely. Most manually-written tests cover only the happy path, leaving boundary conditions, negative paths, and anomalous inputs untested.

Regression bugs slip into production when new code changes miss undocumented edge-case assumptions.

Potential cross-tenant access leaks in multi-room AI routing services go undetected without thorough isolation tests.

Writing boilerplate test scaffolding repeatedly wastes engineering time that should be spent on building features.

Target Users

Software / Backend Developer

Needs a terminal-native assistant to auto-generate high-quality unit tests immediately after writing new business logic in Go.

QA Automation Engineer

Requires a complete edge-case and negative-test reference based on actual code logic to maximize code coverage benchmarks.

Solo Developer / Systems Architect

Manages microservice infrastructure without a dedicated QA team and needs to maintain codebase stability efficiently.

package router

import (
	"errors"
	"strings"
)

var (
	ErrEmptyTenantID    = errors.New("tenant ID cannot be empty")
	ErrUnauthorizedRoom = errors.New("room ID is not in the allowlist")
	ErrPayloadTooLarge  = errors.New("message payload exceeds maximum token limit")
	ErrInvalidPrompt    = errors.New("prompt contains kosong or whitespace only")
)

type DispatchConfig struct {
	AllowedRooms  []string
	MaxPayloadLen int
}

// ValidateDispatch validates a message request before routing to the AI Agent container.
func ValidateDispatch(tenantID, roomID, prompt string, cfg DispatchConfig) (string, error) {
	cleanTenant := strings.TrimSpace(tenantID)
	if cleanTenant == "" {
		return "", ErrEmptyTenantID
	}

	if len(cfg.AllowedRooms) == 0 {
		return "", ErrUnauthorizedRoom
	}

	allowed := false
	for _, r := range cfg.AllowedRooms {
		if r == roomID {
			allowed = true
			break
		}
	}
	if !allowed {
		return "", ErrUnauthorizedRoom
	}

	cleanPrompt := strings.TrimSpace(prompt)
	if cleanPrompt == "" {
		return "", ErrInvalidPrompt
	}

	maxLen := cfg.MaxPayloadLen
	if maxLen <= 0 {
		maxLen = 4096 // default fallback
	}

	if len(cleanPrompt) > maxLen {
		return "", ErrPayloadTooLarge
	}

	return cleanTenant + ":" + roomID, nil
}
# AutoTest-CLI Agent Specification

## Layer 1: Role & Persona
You are a Senior Go QA & Reliability Engineer specializing in
deterministic unit testing, table-driven tests, and edge-case
discovery for lightweight AI agent services.

## Layer 2: Technical Constraints & Standards
- Use ONLY Go's standard `testing` package (`testing`, `errors`, `strings`).
- Do NOT install or import external assertion libraries (like `testify`).
- Enforce idiomatic Go Table-Driven Tests (`[]struct{ name string; ... }`)
  executed with `t.Run()`.
- Never modify the target source file (`validator.go`).

## Layer 3: Analytical & Edge-Case Matrix
For every exported function, cover:
1. Happy Path (valid tenant, allowed room, normal prompt,
   custom & default MaxPayloadLen).
2. Boundary Conditions (whitespace-only strings,
   `MaxPayloadLen <= 0` fallback, exact max length vs `maxLen + 1`).
3. Negative & Security Cases (empty tenant ID, empty
   `AllowedRooms` slice, unauthorized room ID, empty/whitespace prompt).

## Layer 4: Execution & Verification
1. Generate `validator_test.go` in the same directory.
2. Run `go test -v -cover ./...` to verify all test cases pass
   and coverage exceeds 95%.
package router

import (
	"errors"
	"strings"
	"testing"
)

func TestValidateDispatch(t *testing.T) {
	defaultCfg := DispatchConfig{
		AllowedRooms:  []string{"room-1", "room-2"},
		MaxPayloadLen: 100,
	}

	tests := []struct {
		name      string
		tenantID  string
		roomID    string
		prompt    string
		cfg       DispatchConfig
		wantRoute string
		wantErr   error
	}{
		// Happy Path
		{
			name:      "happy path: valid inputs with custom MaxPayloadLen",
			tenantID:  "tenant-abc",
			roomID:    "room-1",
			prompt:    "Hello, agent!",
			cfg:       defaultCfg,
			wantRoute: "tenant-abc:room-1",
			wantErr:   nil,
		},
		{
			name:     "happy path: MaxPayloadLen <= 0 falls back to 4096",
			tenantID: "tenant-fallback",
			roomID:   "room-1",
			prompt:   "A short prompt.",
			cfg: DispatchConfig{
				AllowedRooms:  []string{"room-1"},
				MaxPayloadLen: 0,
			},
			wantRoute: "tenant-fallback:room-1",
			wantErr:   nil,
		},
		// Boundary Conditions
		{
			name:      "boundary: prompt at exact MaxPayloadLen",
			tenantID:  "tenant-boundary",
			roomID:    "room-1",
			prompt:    strings.Repeat("x", 100),
			cfg:       defaultCfg,
			wantRoute: "tenant-boundary:room-1",
			wantErr:   nil,
		},
		{
			name:      "boundary: prompt at maxLen + 1 returns ErrPayloadTooLarge",
			tenantID:  "tenant-abc",
			roomID:    "room-1",
			prompt:    strings.Repeat("y", 101),
			cfg:       defaultCfg,
			wantRoute: "",
			wantErr:   ErrPayloadTooLarge,
		},
		// Negative & Security Cases
		{
			name:      "negative: empty tenant ID",
			tenantID:  "",
			roomID:    "room-1",
			prompt:    "Valid prompt.",
			cfg:       defaultCfg,
			wantRoute: "",
			wantErr:   ErrEmptyTenantID,
		},
		{
			name:      "negative: whitespace-only tenant ID",
			tenantID:  "   ",
			roomID:    "room-1",
			prompt:    "Valid prompt.",
			cfg:       defaultCfg,
			wantRoute: "",
			wantErr:   ErrEmptyTenantID,
		},
		{
			name:      "negative: nil AllowedRooms slice",
			tenantID:  "tenant-abc",
			roomID:    "room-1",
			prompt:    "Valid prompt.",
			cfg: DispatchConfig{
				AllowedRooms:  nil,
				MaxPayloadLen: 100,
			},
			wantRoute: "",
			wantErr:   ErrUnauthorizedRoom,
		},
		{
			name:      "negative: whitespace-only prompt",
			tenantID:  "tenant-abc",
			roomID:    "room-1",
			prompt:    "   ",
			cfg:       defaultCfg,
			wantRoute: "",
			wantErr:   ErrInvalidPrompt,
		},
		// ... 12 more cases covering all logic branches
	}

	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			gotRoute, gotErr := ValidateDispatch(
				tc.tenantID, tc.roomID, tc.prompt, tc.cfg,
			)
			if gotRoute != tc.wantRoute {
				t.Errorf("route: got %q, want %q", gotRoute, tc.wantRoute)
			}
			if !errors.Is(gotErr, tc.wantErr) {
				t.Errorf("error: got %v, want %v", gotErr, tc.wantErr)
			}
		})
	}
}
